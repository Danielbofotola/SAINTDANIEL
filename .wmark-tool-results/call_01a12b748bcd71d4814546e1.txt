240        ]);
241        $interro = DB::table('evaluations')->insertGetId([
242            'affectation_id' => $affectation, 'periode_id' => $periode, 'titre' => 'Interro 1',
243            'type_evaluation' => 'interro', 'max_note' => 20, 'statut' => 'actif',
244            'created_at' => now(), 'updated_at' => now(),

246        $examen = DB::table('evaluations')->insertGetId([

249            'created_at' => now(), 'updated_at' => now(),
250        ]);
251    
252        // 18/20 + 18/20 = 90 % · 10/20 + 10/20 = 50 % · 9,5 + 9,5 = 47,5 % · 8 + 8 = 40 %
