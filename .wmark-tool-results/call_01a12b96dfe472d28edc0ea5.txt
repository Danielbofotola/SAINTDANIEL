1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use App\Models\Classe;
7      use App\Models\FraisScolaire;
8      use App\Models\TypeFrais;
9      use Illuminate\Http\RedirectResponse;
10     use Illuminate\Http\Request;
11     use Illuminate\Validation\Rule;
12     use Illuminate\View\View;
13     
14     class FraisController extends Controller
15     {
16         public function index(): View
17         {
18             return view('admin.frais.index', [
19                 'annee' => AnneeScolaire::enCours(),
20                 'types' => TypeFrais::query()
21                     ->withCount('fraisScolaires')
22                     ->orderBy('nom')
23                     ->get(),
24                 'tarifs' => FraisScolaire::query()
25                     ->anneeEnCours()
26                     ->with(['typeFrais', 'classe'])
27                     ->orderBy('type_frais_id')
28                     ->orderBy('classe_id')
29                     ->get(),
30                 'classes' => Classe::where('statut', 'actif')->orderBy('nom')->get(),
31                 'typesActifs' => TypeFrais::where('statut', 'actif')->orderBy('nom')->get(),
32             ]);
33         }
34     
35         public function storeType(Request $request): RedirectResponse
36         {
37             $data = $request->validate([
38                 'nom' => ['required', 'string', 'max:100', 'unique:types_frais,nom'],
39                 'description' => ['nullable', 'string', 'max:1000'],
40                 'est_inscription' => ['nullable', 'boolean'],
41             ], [
42                 'nom.required' => 'Le nom du type de frais est obligatoire.',
43                 'nom.unique' => 'Ce type de frais existe déjà.',
44             ]);
45     
46             TypeFrais::create($data + ['est_inscription' => $request->boolean('est_inscription'), 'statut' => 'actif']);
47     
48             return redirect()->route('admin.frais.index')->with('success', 'Le type de frais a été créé.');
49         }
50     
51         public function toggleType(TypeFrais $typeFrais): RedirectResponse
52         {
53             $typeFrais->update(['statut' => $typeFrais->statut === 'actif' ? 'inactif' : 'actif']);
54     
55             return back()->with('success', 'Le statut du type de frais a été mis à jour.');
56         }
57     
58         public function destroyType(TypeFrais $typeFrais): RedirectResponse
59         {
60             if ($typeFrais->fraisScolaires()->exists()) {
61                 return back()->with('error', 'Ce type de frais est utilisé par des tarifs : supprimez-les d’abord.');
62             }
63     
64             $typeFrais->delete();
65     
66             return redirect()->route('admin.frais.index')->with('success', 'Le type de frais a été supprimé.');
67         }
68     
69         public function storeTarif(Request $request): RedirectResponse
70         {
71             $annee = AnneeScolaire::enCours();
72     
73             if (! $annee) {
74                 return back()->withInput()->with('error', 'Aucune année scolaire en cours : impossible d’enregistrer un tarif.');
75             }
76     
77             $data = $request->validate([
78                 'type_frais_id' => ['required', Rule::exists('types_frais', 'id')->where('statut', 'actif')],
79                 'classe_id' => ['required', Rule::exists('classes', 'id')->where('statut', 'actif')],
80                 'montant_exige' => ['required', 'numeric', 'min:0'],
81                 'date_echeance' => ['required', 'date'],
82             ], [
83                 'type_frais_id.required' => 'Sélectionnez un type de frais.',
84                 'classe_id.required' => 'Sélectionnez une classe.',
85                 'montant_exige.required' => 'Le montant exigé est obligatoire.',
86                 'date_echeance.required' => 'La date d’échéance est obligatoire.',
87             ]);
88     
89             $exists = FraisScolaire::query()
90                 ->where('type_frais_id', $data['type_frais_id'])
91                 ->where('classe_id', $data['classe_id'])
92                 ->where('annee_scolaire_id', $annee->id)
93                 ->exists();
94     
95             if ($exists) {
96                 return back()->withInput()->with('error', 'Un tarif existe déjà pour ce type de frais et cette classe cette année.');
97             }
98     
99             FraisScolaire::create($data + ['annee_scolaire_id' => $annee->id, 'statut' => 'actif']);
100    
101            return redirect()->route('admin.frais.index')->with('success', 'Le tarif a été enregistré.');
102        }
103    
104        public function destroyTarif(FraisScolaire $fraisScolaire): RedirectResponse
105        {
106            if ($fraisScolaire->paiements()->exists()) {
107                return back()->with('error', 'Ce tarif a déjà des paiements enregistrés : supprimez-les d’abord.');
108            }
109    
110            $fraisScolaire->delete();
111    
112            return redirect()->route('admin.frais.index')->with('success', 'Le tarif a été supprimé.');
113        }
114    }
