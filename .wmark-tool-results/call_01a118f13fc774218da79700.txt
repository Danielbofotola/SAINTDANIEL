61     function sessionActive(string $id, ?int $userId, string $ip, string $agent, int $minutesAgo = 0): void
62     {
63         DB::table('sessions')->insert([
64             'id' => $id,
65             'user_id' => $userId,
66             'ip_address' => $ip,
67             'user_agent' => $agent,
68             'payload' => base64_encode(serialize([])),
69             'last_activity' => now()->subMinutes($minutesAgo)->getTimestamp(),
70         ]);
71     }

73     test('the admin sees the accounts currently online, not the stale sessions', function () {
74         $admin = User::factory()->create(['role' => 'admin', 'username' => 'admin.test']);
75         $comptable = User::factory()->create(['role' => 'comptable', 'username' => 'comptable.test']);
76     
77         sessionActive('sess-admin', $admin->id, '10.0.0.1', 'Mozilla/5.0 (Windows NT 10.0) Chrome/120.0', 0);
78         sessionActive('sess-vieux', $comptable->id, '10.0.0.2', 'Mozilla/5.0 Firefox/121.0', 40);
79         sessionActive('sess-invite', null, '10.0.0.3', 'Mozilla/5.0 (iPhone; CPU iPhone OS 17_0) Safari', 0);
80     
81         $this->actingAs($admin)
82             ->get(route('admin.profile.online'))
83             ->assertOk()
84             ->assertSee('Utilisateurs en ligne')
85             ->assertSee('admin.test')
86             ->assertSee('10.0.0.1')
87             ->assertSee('Chrome · Windows')
88             ->assertDontSee('comptable.test')          // session trop ancienne
89             ->assertDontSee('10.0.0.2')
90             ->assertSee('Sessions invitées')           // session sans compte
91             ->assertSee('Safari · iPhone');
92     
93         expect(PresenceUtilisateurs::idsEnLigne())->toBe([$admin->id]);
94     
95         // Le profil administrateur donne accès à la page.
96         $this->actingAs($admin)
97             ->get(route('admin.profile.show'))
98             ->assertOk()
99             ->assertSee(route('admin.profile.online'), false);
100    });
