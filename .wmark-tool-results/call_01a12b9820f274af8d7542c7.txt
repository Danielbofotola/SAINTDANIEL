3      namespace App\Models;
4      
5      use App\Models\Concerns\BelongsToAnneeScolaire;
6      use Illuminate\Database\Eloquent\Model;
7      
8      class FraisScolaire extends Model
9      {

15     
16         protected function casts(): array
17         {
18             return [
19                 'montant_exige' => 'decimal:2',
20                 'date_echeance' => 'date',
21             ];
22         }
23     
24         public function typeFrais()
