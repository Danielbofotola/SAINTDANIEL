1      <?php
2      
3      namespace App\Providers;
4      
5      use App\Models\ParametreSysteme;
6      use App\Models\AnneeScolaire;
7      use Illuminate\Database\Eloquent\Relations\Relation;
8      use Illuminate\Support\ServiceProvider;
9      use Illuminate\Support\Facades\View;
10     
11     class AppServiceProvider extends ServiceProvider
12     {
13         /**
14          * Register any application services.
15          */
16         public function register(): void
17         {
18             //
19         }
20     
21         /**
22          * Bootstrap any application services.
23          */
24         public function boot(): void
25         {
26             // Bénéficiaires d'un salaire : un professeur (entité) ou un autre
27             // membre du personnel (compte utilisateur).
28             Relation::morphMap([
29                 'professeur' => \App\Models\Professeurs::class,
30                 'utilisateur' => \App\Models\User::class,
31             ]);
32     
33             View::composer('*', static function ($view): void {
34                 $settings = request()->attributes->get('system_settings');
35                 if (! $settings) {
36                     $settings = self::firstOrNull(fn () => ParametreSysteme::query()->first());
37                     $settings ??= ParametreSysteme::defaults();
38                     request()->attributes->set('system_settings', $settings);
39                 }
40     
41                 $view->with('systemSettings', $settings);
42     
43                 if (! request()->attributes->has('annee_en_cours')) {
44                     request()->attributes->set('annee_en_cours', self::firstOrNull(
45                         fn () => AnneeScolaire::where('is_encours', true)->first()
46                     ));
47                 }
48     
49                 $view->with('anneeEnCours', request()->attributes->get('annee_en_cours'));
50             });
51         }
52     
53         /**
54          * Exécute une lecture destinée à l'affichage et renvoie null en cas d'échec.
55          *
56          * Ces valeurs ne sont que du contexte d'affichage : une table absente ou
57          * indisponible ne doit jamais faire échouer le rendu d'une page.
58          */
59         private static function firstOrNull(callable $query): mixed
60         {
61             try {
62                 return $query();
63             } catch (\Throwable) {
64                 return null;
65             }
66         }
67     }
