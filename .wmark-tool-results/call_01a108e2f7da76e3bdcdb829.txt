1      <?php
2      
3      namespace App\Models;
4      
5      use Illuminate\Database\Eloquent\Model;
6      
7      class Professeurs extends Model
8      {
9          protected $fillable = ['matricule', 'nom', 'postnom', 'prenom', 'sexe', 'specialite', 'telephone', 'email', 'photo', 'statut'];
10     
11         public function user()
12         {
13             return $this->hasOne(User::class, 'professeur_id');
14         }
15     
16         public function affectationsCours()
17         {
18             return $this->hasMany(AffectationCours::class, 'professeur_id');
19         }
20     
21         public function documents()
22         {
23             return $this->hasMany(DocumentProfesseur::class, 'professeur_id');
24         }
25     
26         public function getNomCompletAttribute(): string
27         {
28             return trim($this->nom.' '.$this->postnom.' '.$this->prenom);
29         }
30     }
