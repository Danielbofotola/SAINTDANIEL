87     test('every role gets a messenger button on a page that does not offer it', function () {
88         foreach (['professeur', 'comptable', 'prefetture', 'titulaire', 'admin'] as $role) {
89             $compte = User::factory()->create(['role' => $role]);
90     
91             $html = $this->actingAs($compte)
92                 ->get(route('notifications.index'))
93                 ->assertOk()
94                 ->getContent();
95     
96             // Le bouton est injecté une seule fois, avec le lien vers la messagerie.
97             expect(substr_count($html, '<a class="messagerie-flottante'))->toBe(1)
98                 ->and($html)->toContain(route('messagerie.index'))
99                 ->and($html)->toContain('aria-label="Messagerie"');
100        }
101    });

103    test('the button carries the unread counter', function () {
104        $alice = User::factory()->create(['role' => 'admin']);
105        $bruno = User::factory()->create(['role' => 'comptable']);
106    
107        App\Support\Messagerie::creer($alice, [$bruno->id], 'Question', 'Bonjour Bruno.');
108    
109        $html = $this->actingAs($bruno)->get(route('notifications.index'))->assertOk()->getContent();
110    
111        expect($html)->toContain('data-non-lus')
112            ->and(substr_count($html, '<a class="messagerie-flottante'))->toBe(1);
113    
114        // Une fois la messagerie consultée, le compteur disparaît du bouton.
115        App\Support\Messagerie::marquerLu(App\Support\Messagerie::conversationsDe($bruno)->first(), $bruno);
116    
117        $html = $this->actingAs($bruno)->get(route('notifications.index'))->assertOk()->getContent();
118    
119        expect($html)->toContain('<a class="messagerie-flottante')
120            ->not->toContain('data-non-lus');
121    });

123    test('no duplicate on a page that already offers the messenger', function () {
124        $compte = User::factory()->create(['role' => 'comptable']);
125    
126        // La page /messagerie contient déjà ses propres liens : pas d'injection.
127        $html = $this->actingAs($compte)->get(route('messagerie.index'))->assertOk()->getContent();
