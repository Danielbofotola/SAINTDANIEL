580                donnees.append('sujet', sujet);
581                donnees.append('corps', corps);
582                choix.forEach(c => donnees.append('destinataires[]', c));
583                const rep = await fetch(STORE_URL, {
584                    method: 'POST',
585                    headers: { 'X-Requested-With': 'XMLHttpRequest', 'Accept': 'application/json' },

588                });

591                window.location.assign(ESPACE_URL + '?conversation=' + json.conversation_id);
592            } catch (e) {
593                alert('Envoi impossible. Vérifiez le destinataire.');

595                this.disabled = false;
596            }
597        });
