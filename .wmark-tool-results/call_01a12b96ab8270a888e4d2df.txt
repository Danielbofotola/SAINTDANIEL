121    <script>
122        (function () {
123            var payeParPaire = @json($soldePaye);
124            var montantsFrais = @json($montantsFrais);
125            var classesParInscription = @json($classesParInscription ?? []);
126            var classesParFrais = @json($classesParFrais ?? []);
127            var tarifsDisponibles = @json($tarifsDisponibles ?? []);
128            var nomsClasses = @json($nomsClasses ?? []);
129            var recherche = document.getElementById('eleveSearch');
130            var inscriptionSelect = document.getElementById('inscriptionSelect');
131            var fraisSelect = document.getElementById('fraisSelect');
132            var soldeInfo = document.getElementById('soldeInfo');
133    
134            function filtrerEleves() {
135                if (!recherche || !inscriptionSelect) { return; }
136                var terme = recherche.value.trim().toLowerCase();
137                Array.prototype.forEach.call(inscriptionSelect.options, function (option) {
138                    if (!option.value) { option.hidden = false; return; }
139                    option.hidden = terme !== '' && option.text.toLowerCase().indexOf(terme) === -1;
140                });
141            }
142    
143            function majSolde() {
144                if (!inscriptionSelect || !fraisSelect || !soldeInfo) { return; }
145                var inscription = inscriptionSelect.value;
146                var frais = fraisSelect.value;
147                if (!inscription || !frais) { soldeInfo.classList.add('d-none'); return; }
148                var exige = Number(montantsFrais[frais] || 0);
149                var paye = Number(payeParPaire[inscription + '-' + frais] || 0);
150                var reste = Math.max(exige - paye, 0);
151                soldeInfo.classList.remove('d-none');
152                soldeInfo.innerHTML = 'Montant exigé : <b>' + exige.toFixed(2) + '</b> · Déjà payé : <b>' + paye.toFixed(2) + '</b> · <b>Reste à payer : ' + reste.toFixed(2) + '</b>';
153            }
154    
155            // Reconstruit la liste des tarifs : ne garde QUE ceux de la classe de
156            // l'élève sélectionné (les tarifs des autres classes sont retirés du DOM,
157            // et non simplement masqués).
158            function rendreTarifs() {
159                if (!fraisSelect) { return; }
160    
161                var inscription = inscriptionSelect ? inscriptionSelect.value : '';
162                var classe = inscription ? String(classesParInscription[inscription] || '') : '';
163                var courant = fraisSelect.value;
164                var disponibles = tarifsDisponibles.filter(function (tarif) {
165                    return classe === '' || String(tarif.classe) === classe;
166                });
167    
168                var label = document.getElementById('tarifLabel');
169                if (label) {
170                    label.textContent = (classe !== '' && nomsClasses[classe])
171                        ? 'Tarif * (classe ' + nomsClasses[classe] + ')'
172                        : "Tarif de la classe de l'élève *";
173                }
174    
175                fraisSelect.innerHTML = '';
176                var vide = document.createElement('option');
177                vide.value = '';
178                vide.textContent = classe !== '' && nomsClasses[classe]
179                    ? 'Choisir un tarif de ' + nomsClasses[classe]
180                    : 'Choisir un tarif';
181                fraisSelect.appendChild(vide);
182    
183                disponibles.forEach(function (tarif) {
184                    var option = document.createElement('option');
185                    option.value = tarif.id;
186                    option.textContent = tarif.libelle;
187                    fraisSelect.appendChild(option);
188                });
189    
190                var encoreValide = disponibles.some(function (tarif) { return String(tarif.id) === String(courant); });
191                if (encoreValide) {
192                    fraisSelect.value = courant;
193                } else if (disponibles.length === 1) {
194                    fraisSelect.value = disponibles[0].id;
195                }
196    
197                majSolde();
198            }
199    
200            if (recherche) { recherche.addEventListener('input', filtrerEleves); }
201            if (inscriptionSelect) { inscriptionSelect.addEventListener('change', rendreTarifs); }
202            if (fraisSelect) { fraisSelect.addEventListener('change', majSolde); }
203            rendreTarifs();
204        })();
205    </script>
