1      <?php

3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use App\Models\Classe;
7      use App\Models\DemandeCompte;
8      use App\Models\DemandeInscription;
9      use Illuminate\Http\RedirectResponse;
10     use Illuminate\Http\Request;
11     use Illuminate\Support\Facades\Hash;
12     use Illuminate\Validation\Rule;
13     use Illuminate\View\View;
14     
15     class RegistrationRequestController extends Controller
16     {
17         public function createStudentRequest(): View
18         {
19             return view('registration.student-request', [
20                 'classes' => Classe::query()
21                     ->where('statut', 'actif')
22                     ->whereHas('cycle', fn ($query) => $query->where('statut', 'actif'))
23                     ->orderBy('nom')
24                     ->get(),
25                 'schoolYears' => AnneeScolaire::query()
26                     ->where('statut', 'actif')
27                     ->orderByDesc('is_encours')
28                     ->orderByDesc('date_debut')
29                     ->get(),
30             ]);
31         }
32     
33         public function storeStudentRequest(Request $request): RedirectResponse
34         {
35             $data = $request->validate([
36                 'nom' => ['required', 'string', 'max:50'],
37                 'postnom' => ['nullable', 'string', 'max:50'],
38                 'prenom' => ['required', 'string', 'max:50'],
39                 'sexe' => ['required', 'in:M,F'],
40                 'date_naissance' => ['required', 'date', 'before:today'],
41                 'lieu_naissance' => ['nullable', 'string', 'max:100'],
42                 'classe_id' => ['required', Rule::exists('classes', 'id')->where('statut', 'actif')],
43                 'annee_scolaire_id' => ['required', Rule::exists('annees_scolaires', 'id')->where('statut', 'actif')],
44                 'nom_mere' => ['required', 'string', 'max:100'],
45                 'telephone_mere' => ['required', 'string', 'max:20'],
46                 'email_mere' => ['nullable', 'email', 'max:100'],
47                 'adresse_mere' => ['nullable', 'string', 'max:1000'],
48                 'nom_pere' => ['nullable', 'string', 'max:100'],
49                 'telephone_pere' => ['nullable', 'string', 'max:20'],
50                 'email_pere' => ['nullable', 'email', 'max:100'],
51             ], [
52                 'date_naissance.before' => 'La date de naissance doit être antérieure à aujourd’hui.',
53                 'classe_id.exists' => 'La classe sélectionnée n’est plus disponible.',
54                 'annee_scolaire_id.exists' => 'L’année scolaire sélectionnée n’est plus disponible.',
55             ]);
56     
57             DemandeInscription::create($data);
58     
59             return redirect()->route('registration.student.create')
60                 ->with('success', 'Votre demande a été transmise. Elle sera examinée par l’administration avant toute inscription définitive.');
61         }
62     
63         public function createAccountRequest(): View
64         {
65             return view('registration.account-request');
66         }
67     
68         public function storeAccountRequest(Request $request): RedirectResponse
69         {
70             $data = $request->validate([
71                 'nom' => ['required', 'string', 'max:50'],
72                 'postnom' => ['nullable', 'string', 'max:50'],
73                 'prenom' => ['required', 'string', 'max:50'],
74                 'username' => [
75                     'required', 'string', 'max:50',
76                     Rule::unique('users', 'username'),
77                     Rule::unique('demandes_comptes', 'username')->where('statut', 'en_attente'),
78                 ],
79                 'email' => [
80                     'required', 'email', 'max:100',
81                     Rule::unique('users', 'email'),
82                     Rule::unique('demandes_comptes', 'email')->where('statut', 'en_attente'),
83                 ],
84                 'telephone' => ['nullable', 'string', 'max:20'],
85                 'role_souhaite' => ['required', 'in:prefetture,titulaire,professeur,comptable'],
86                 'mot_de_passe' => ['required', 'string', 'min:8', 'confirmed'],
87             ], [
88                 'username.unique' => 'Ce nom d’utilisateur est déjà utilisé ou fait l’objet d’une demande en attente.',
89                 'email.unique' => 'Cette adresse e-mail est déjà utilisée ou fait l’objet d’une demande en attente.',
90                 'role_souhaite.in' => 'Le rôle demandé n’est pas disponible.',
91                 'mot_de_passe.min' => 'Le mot de passe doit contenir au moins 8 caractères.',
92                 'mot_de_passe.confirmed' => 'La confirmation du mot de passe ne correspond pas.',
93             ]);
94     
95             $data['mot_de_passe'] = Hash::make($data['mot_de_passe']);
96             DemandeCompte::create($data);
97     
98             return redirect()->route('registration.account.create')
99                 ->with('success', 'Votre demande de compte a été transmise. Vous pourrez vous connecter après son approbation par l’administration.');
100        }
101    }
