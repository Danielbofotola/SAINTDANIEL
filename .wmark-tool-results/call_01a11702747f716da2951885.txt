79     test('a user writes to a colleague who then sees the message', function () {
80         $alice = User::factory()->create(['role' => 'admin', 'username' => 'alice']);
81         $bruno = User::factory()->create(['role' => 'professeur', 'username' => 'bruno']);
82     
83         $this->actingAs($alice)
84             ->get(route('messagerie.index'))
85             ->assertOk()
86             ->assertSee('Messagerie interne')
87             ->assertSee('Nouveau message');
88     
89         $this->actingAs($alice)
90             ->post(route('messagerie.store'), [
91                 'destinataires' => [(string) $bruno->id],
92                 'sujet' => 'Préparation des bulletins',
93                 'corps' => 'Peux-tu préparer les moyennes du 1er trimestre ?',
94             ])
95             ->assertRedirect()
96             ->assertSessionHas('success', fn (string $m) => str_contains($m, '1 destinataire'));
97     
98         $conversation = Conversation::first();
99         expect($conversation->type)->toBe('directe')
100            ->and($conversation->participants()->count())->toBe(2)
101            ->and(Messagerie::nonLus($bruno))->toBe(1)      // non lu pour Bruno
102            ->and(Messagerie::nonLus($alice))->toBe(0);     // l'auteur a lu son propre message
103    
104        $this->actingAs($bruno)
105            ->get(route('messagerie.index', ['conversation' => $conversation->id]))
106            ->assertOk()
107            ->assertSee('Peux-tu préparer les moyennes');   // le message s'affiche dans le fil
108    
109        // La consultation marque la conversation comme lue.
110        expect(Messagerie::nonLus($bruno->fresh()))->toBe(0);
111    });
