130        <script>
131            const sectionSelect = document.getElementById('section_id');
132            const optionSelect = document.getElementById('option_id');
133            const createOptionLink = document.getElementById('create-option-link');
134            const options = Array.from(optionSelect.options);
135    
136            function filterOptions() {
137                const sectionId = sectionSelect.value;
138                options.forEach((option, index) => {
139                    if (index === 0) return;
140                    const visible = Boolean(sectionId) && option.dataset.sectionId === sectionId;
141                    option.hidden = !visible;
142                    option.disabled = !visible;
143                    if (!visible && option.selected) option.selected = false;
144                });
145            }
146    
147            sectionSelect.addEventListener('change', filterOptions);
148            sectionSelect.addEventListener('change', function () {
149                const optionUrl = new URL(createOptionLink.href);
150                if (sectionSelect.value) {
151                    optionUrl.searchParams.set('section_id', sectionSelect.value);
152                } else {
153                    optionUrl.searchParams.delete('section_id');
154                }
155                createOptionLink.href = optionUrl.toString();
156            });
157            filterOptions();
158        </script>
