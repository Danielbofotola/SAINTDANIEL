1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\Annonce;
6      use App\Models\ParametreSysteme;
7      use App\Models\User;
8      use App\Support\Messagerie;
9      use Illuminate\Http\Request;
10     use Illuminate\View\View;
11     
12     /**
13      * Espace Parent : page de communication dédiée aux parents, ouverte seulement
14      * quand l'administrateur active le module « Portail Parents ». Elle réutilise
15      * la messagerie interne (conversations, messages, pièces jointes dont les
16      * vocaux) avec une interface grand public.
17      */
18     class PortailParentController extends Controller
19     {
20         public function index(Request $request): View
21         {
22             $utilisateur = $request->user();
23             $parametres = ParametreSysteme::current();
24     
25             // Le portail n'existe que si l'administration l'a ouvert.
26             abort_unless($parametres->portail_parents_actif, 404);
27     
28             $conversations = Messagerie::conversationsDe($utilisateur);
29     
30             $active = $request->query('conversation')
31                 ? $conversations->firstWhere('id', (int) $request->query('conversation'))
32                 : $conversations->first();
33     
34             if ($active) {
35                 Messagerie::marquerLu($active, $utilisateur);
36                 $active->setAttribute('non_lus', 0);
37             }
38     
39             return view('parent.portail', [
40                 'user' => $utilisateur,
41                 'parametres' => $parametres,
42                 'conversations' => $conversations,
43                 'active' => $active,
44                 'messages' => $active ? $active->messages()->with('auteur')->get() : collect(),
45                 'annonces' => Annonce::query()
46                     ->where(fn ($query) => $query->where('cible', 'tous')->orWhere('cible', 'parent'))
47                     ->latest()
48                     ->limit(10)
49                     ->get(),
50                 'notifications' => $utilisateur->notifications()->latest()->limit(15)->get(),
51                 // Correspondants possibles : le personnel de l'établissement.
52                 'destinataires' => User::query()
53                     ->where('statut', 'actif')
54                     ->where('role', '!=', 'parent')
55                     ->where('id', '!=', $utilisateur->id)
56                     ->orderBy('nom')
57                     ->orderBy('prenom')
58                     ->get(),
59                 'nonLus' => (int) $conversations->sum('non_lus'),
60             ]);
61         }
62     }
