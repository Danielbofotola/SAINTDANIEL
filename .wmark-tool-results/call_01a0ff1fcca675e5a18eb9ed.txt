1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use Illuminate\Http\RedirectResponse;
7      use Illuminate\Http\Request;
8      use Illuminate\Support\Facades\DB;
9      use Illuminate\View\View;
10     
11     class AnneeScolaireController extends Controller
12     {
13         /**
14          * Affiche le formulaire de création et la liste des années scolaires.
15          */
16         public function index(): View
17         {
18             return view('admin.annees.index', [
19                 'annees' => AnneeScolaire::orderByDesc('date_debut')
20                     ->orderByDesc('libelle')
21                     ->get(),
22             ]);
23         }
24     
25         /**
26          * Enregistre une nouvelle année scolaire.
27          */
28         public function store(Request $request): RedirectResponse
29         {
30             $data = $request->validate([
31                 'libelle' => ['required', 'string', 'max:20', 'unique:annees_scolaires,libelle'],
32                 'date_debut' => ['required', 'date'],
33                 'date_fin' => ['required', 'date', 'after:date_debut'],
34             ]);
35     
36             $data['statut'] = 'actif';
37             $data['is_encours'] = false;
38     
39             AnneeScolaire::create($data);
40     
41             return redirect()
42                 ->route('admin.annees.index')
43                 ->with('success', "L'année scolaire « {$data['libelle']} » a été créée avec succès.");
44         }
45     
46         /**
47          * Active / désactive une année scolaire.
48          *
49          * Une seule année peut être « en cours » à la fois : activer une année
50          * désactive automatiquement les autres.
51          */
52         public function toggle(AnneeScolaire $annee): RedirectResponse
53         {
54             $activation = ! $annee->is_encours;
55     
56             DB::transaction(function () use ($annee, $activation) {
57                 AnneeScolaire::query()->update(['is_encours' => false]);
58                 $annee->update(['is_encours' => $activation]);
59             });
60     
61             return back()->with(
62                 'success',
63                 $activation
64                     ? "L'année « {$annee->libelle} » est désormais l'année en cours."
65                     : "L'année « {$annee->libelle} » n'est plus l'année en cours."
66             );
67         }
68     }
