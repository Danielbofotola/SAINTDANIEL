1      <?php
2      
3      namespace App\Models;
4      
5      use Illuminate\Database\Eloquent\Model;
6      
7      /**
8       * Décision de fin d'année pour une inscription : la moyenne vient du bulletin,
9       * la décision (admis / ajourné / redouble) du jury et de ses critères.
10      */
11     class Deliberation extends Model
12     {
13         public const DECISIONS = [
14             'admis' => 'Admis (passe)',
15             'ajourne' => 'Ajourné (repêchage)',
16             'redouble' => 'Redoublant',
17         ];
18     
19         protected $fillable = [
20             'inscription_id', 'annee_scolaire_id', 'pourcentage_general', 'cours_echoues',
21             'decision', 'critere', 'manuelle', 'classe_suivante_id', 'decide_par', 'decide_le', 'commentaire',
22         ];
23     
24         protected function casts(): array
25         {
26             return [
27                 'pourcentage_general' => 'decimal:2',
28                 'manuelle' => 'boolean',
29                 'decide_le' => 'datetime',
30             ];
31         }
32     
33         public function inscription()
34         {
35             return $this->belongsTo(Inscription::class);
36         }
37     
38         public function anneeScolaire()
39         {
40             return $this->belongsTo(AnneeScolaire::class);
41         }
42     
43         public function classeSuivante()
44         {
45             return $this->belongsTo(Classe::class, 'classe_suivante_id');
46         }
47     
48         public function decidePar()
49         {
50             return $this->belongsTo(User::class, 'decide_par');
51         }
52     
53         public function libelleDecision(): string
54         {
55             return self::DECISIONS[$this->decision] ?? $this->decision;
56         }
57     }
