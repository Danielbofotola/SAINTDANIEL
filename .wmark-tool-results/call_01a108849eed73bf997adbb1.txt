1      <?php
2      
3      namespace App\Models;
4      
5      use Illuminate\Database\Eloquent\Model;
6      use Illuminate\Support\Facades\DB;
7      use Illuminate\Support\Facades\Schema;
8      
9      class AnneeScolaire extends Model
10     {
11         /**
12          * Tables portant une clé étrangère vers annees_scolaires.
13          */
14         private const TABLES_LIEES = [
15             'inscriptions',
16             'periodes',
17             'affectations_cours',
18             'frais_scolaires',
19             'depenses',
20             'demandes_inscription',
21         ];
22     
23         protected $table = 'annees_scolaires';
24         protected $fillable = ['libelle', 'date_debut', 'date_fin', 'is_encours', 'statut'];
25     
26         protected function casts(): array
27         {
28             return [
29                 'date_debut' => 'date',
30                 'date_fin' => 'date',
31                 'is_encours' => 'boolean',
32             ];
33         }
34     
35         /**
36          * L'année scolaire « en cours » (unique), ou null si aucune n'est définie.
37          */
38         public static function enCours(): ?self
39         {
40             return static::where('is_encours', true)->first();
41         }
42     
43         /**
44          * Les inscriptions rattachées à cette année scolaire.
45          */
46         public function inscriptions()
47         {
48             return $this->hasMany(Inscription::class);
49         }
50     
51         public function periodes()
52         {
53             return $this->hasMany(Periode::class);
54         }
55     
56         public function affectationsCours()
57         {
58             return $this->hasMany(AffectationCours::class);
59         }
60     
61         public function fraisScolaires()
62         {
63             return $this->hasMany(FraisScolaire::class);
64         }
65     
66         public function depenses()
67         {
68             return $this->hasMany(Depense::class);
69         }
70     
71         public function archive()
72         {
73             return $this->hasOne(ArchiveAnnee::class);
74         }
75     
76         /**
77          * Indique si l'année est référencée par des données métier
78          * (inscriptions, périodes, finances…), ce qui interdit sa suppression.
79          */
80         public function isUtilisee(): bool
81         {
82             foreach (self::TABLES_LIEES as $table) {
83                 if (Schema::hasTable($table) && DB::table($table)->where('annee_scolaire_id', $this->id)->exists()) {
84                     return true;
85                 }
86             }
87     
88             return false;
89         }
90     }
