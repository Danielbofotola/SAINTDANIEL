238    test('désactiver un élève le rend invisible et l’activation le rétablit', function () {
239        $eleveId = seedEleve();
240        $admin = User::factory()->create(['role' => 'admin']);
241    
242        // Désactivation : l'élève sort des effectifs et des écrans de suivi.
243        $this->actingAs($admin)->patch(route('admin.eleves.toggle', $eleveId))
244            ->assertRedirect()
245            ->assertSessionHas('success', fn ($m) => str_contains($m, 'désactivé'));
246    
247        expect(\App\Models\Eleve::find($eleveId)->statut)->toBe('inactif')
248            ->and(\App\Models\Eleve::inscritAnneeEnCours()->whereKey($eleveId)->exists())->toBeFalse();
249    
250        // Son dossier reste consultable : le suivi administratif continue.
251        $this->actingAs($admin)->get(route('admin.eleves.show', $eleveId))->assertOk()
252            ->assertSee("Dossier de l'élève", false)
253            ->assertSee('Responsables légaux');
254    
255        // Réactivation : il réapparaît dans les listes et les effectifs.
256        $this->actingAs($admin)->patch(route('admin.eleves.toggle', $eleveId))->assertRedirect();
