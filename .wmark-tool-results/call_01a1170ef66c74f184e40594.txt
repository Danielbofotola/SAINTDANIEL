71     test('every role sees the floating bell with the unread count and the announcements', function () {
72         $admin = User::factory()->create(['role' => 'admin']);
73         $comptable = User::factory()->create(['role' => 'comptable', 'username' => 'comptable.test']);
74         $prefet = User::factory()->create(['role' => 'prefetture', 'username' => 'prefet.test']);
75         $titulaire = User::factory()->create(['role' => 'titulaire', 'username' => 'titulaire.test']);
76     
77         $this->actingAs($admin)->post(route('admin.annonces.store'), [
78             'titre' => 'Réunion pédagogique vendredi',
79             'message' => 'Présence obligatoire de tout le personnel à 14 h.',
80             'niveau' => 'warning',
81             'cible' => 'tous',
82         ]);
83     
84         foreach ([$comptable, $prefet, $titulaire] as $compte) {
85             Auth::login($compte);
86             $cloche = view('partials.cloche-flottante')->render();
87     
88             expect($cloche)->toContain('cloche-flottante')
89                 ->toContain('aria-label="Notifications"')
90                 ->toContain('Réunion pédagogique vendredi')      // le titre de l'annonce
91                 ->toContain('data-non-lues')                     // le compteur de non lues
92                 ->toContain(route('notifications.index'));       // le lien vers la boîte de réception
93     
94             // La boîte de réception est accessible et affiche l'annonce.
95             $this->actingAs($compte)
96                 ->get(route('notifications.index'))
97                 ->assertOk()
98                 ->assertSee('Réunion pédagogique vendredi')
99                 ->assertSee('Nouveau');
100        }
101    });

103    test('the bell is hidden once everything is read', function () {
104        $admin = User::factory()->create(['role' => 'admin']);
105        $comptable = User::factory()->create(['role' => 'comptable']);
106    
107        $this->actingAs($admin)->post(route('admin.annonces.store'), [

115    
116        $sansBadge = view('partials.cloche-flottante')->render();
