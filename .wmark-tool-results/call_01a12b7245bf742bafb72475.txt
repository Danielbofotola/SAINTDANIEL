1      <?php
2      
3      namespace App\Support;
4      
5      use App\Models\Inscription;
6      use App\Models\ParametreSysteme;
7      
8      /**
9       * Jury de délibération : applique les critères de l'établissement à la moyenne
10      * du bulletin. Il ne recalcule JAMAIS la moyenne lui-même — il consomme
11      * `BulletinRdc`, pour que la délibération et le bulletin restent d'accord.
12      */
13     class Jury
14     {
15         public function __construct(private readonly ParametreSysteme $regles)
16         {
17         }
18     
19         public static function regles(): self
20         {
21             return new self(ParametreSysteme::current());
22         }
23     
24         /**
25          * Analyse d'une inscription : moyenne du bulletin, branches en échec, et
26          * décision proposée par les critères.
27          *
28          * @return array{pourcentage: float, cours_echoues: int, branches_repechage: array<int, string>, decision_proposee: string}
29          */
30         public function analyser(Inscription $inscription): array
31         {
32             $bulletin = BulletinRdc::for($inscription)->build();
33     
34             $pourcentage = (float) $bulletin['pourcentage_general'];
35             $branchesEchouees = $bulletin['branches_repechage'] ?? [];
36             $coursEchoues = count($branchesEchouees);
37     
38             return [
39                 'pourcentage' => $pourcentage,
40                 'cours_echoues' => $coursEchoues,
41                 'branches_repechage' => $branchesEchouees,
42                 'decision_proposee' => $this->decider($pourcentage, $coursEchoues),
43             ];
44         }
45     
46         /** Décision selon les critères : seuil d'admission, de repêchage, plafond de cours en échec. */
47         public function decider(float $pourcentage, int $coursEchoues): string
48         {
49             $admission = (float) ($this->regles->deliberation_seuil_admission ?? 50);
50             $repechage = (float) ($this->regles->deliberation_seuil_repechage ?? 45);
51             $plafond = $this->regles->deliberation_max_cours_echoues;
52     
53             $tropDeCoursEchoues = $plafond !== null && $coursEchoues > (int) $plafond;
54     
55             if ($pourcentage >= $admission && ! $tropDeCoursEchoues) {
56                 return 'admis';
57             }
58     
59             return $pourcentage >= $repechage ? 'ajourne' : 'redouble';
60         }
61     
62         /** Photo des critères appliqués, conservée avec la décision. */
63         public function critere(): array
64         {
65             return [
66                 'seuil_admission' => (float) ($this->regles->deliberation_seuil_admission ?? 50),
67                 'seuil_repechage' => (float) ($this->regles->deliberation_seuil_repechage ?? 45),
68                 'max_cours_echoues' => $this->regles->deliberation_max_cours_echoues,
69                 'note_min_cours' => $this->regles->deliberation_note_min_cours,
70             ];
71         }
72     }
