15         public function index(Request $request): View
16         {
17             $search = trim((string) $request->query('q', ''));
18             $statut = (string) $request->query('statut', '');
19     
20             $professeurs = Professeurs::query()
21                 ->withCount(['affectationsCours as affectations_count' => fn ($query) => $query
22                     ->anneeEnCours()
23                     ->where('statut', 'actif')])
24                 ->when($search !== '', fn ($query) => $query->where(function ($query) use ($search) {
25                     $query->where('nom', 'like', "%{$search}%")
26                         ->orWhere('postnom', 'like', "%{$search}%")
27                         ->orWhere('prenom', 'like', "%{$search}%")
28                         ->orWhere('matricule', 'like', "%{$search}%")
29                         ->orWhere('specialite', 'like', "%{$search}%")
30                         ->orWhere('email', 'like', "%{$search}%");
31                 }))
32                 ->when(in_array($statut, ['actif', 'inactif'], true), fn ($query) => $query->where('statut', $statut))
33                 ->orderBy('nom')
34                 ->orderBy('postnom')
35                 ->orderBy('prenom')
36                 ->paginate(25)
37                 ->withQueryString();
38     
39             return view('admin.professeurs.index', [
40                 'professeurs' => $professeurs,
41                 'search' => $search,
42                 'statut' => $statut,
43                 'stats' => [
44                     'total' => Professeurs::count(),
45                     'actifs' => Professeurs::where('statut', 'actif')->count(),
46                     'inactifs' => Professeurs::where('statut', 'inactif')->count(),
47                 ],
48             ]);
49         }

69         public function emploiDuTemps(Professeurs $professeur): View
70         {
71             $entrees = EmploiDuTemps::query()
72                 ->with(['affectation.cours', 'affectation.classe', 'salle'])
73                 ->whereHas('affectation', fn ($query) => $query
74                     ->where('professeur_id', $professeur->id)
75                     ->anneeEnCours())
76                 ->orderBy('heure_debut')
77                 ->get();
78     
79             $creneaux = $entrees
80                 ->map(fn ($entree) => ['debut' => $entree->heure_debut, 'fin' => $entree->heure_fin])
81                 ->unique(fn ($creneau) => $creneau['debut'].'|'.$creneau['fin'])
82                 ->sortBy('debut')
83                 ->values();
84     
85             $grille = [];
86             foreach ($entrees as $entree) {
87                 $grille[$entree->jour][$entree->heure_debut.'|'.$entree->heure_fin][] = $entree;
88             }
89     
90             $totalMinutes = $entrees->sum(fn ($entree) => (strtotime($entree->heure_fin) - strtotime($entree->heure_debut)) / 60);
91     
92             return view('admin.professeurs.emploi-du-temps', [
93                 'professeur' => $professeur,
94                 'jours' => ['lundi', 'mardi', 'mercredi', 'jeudi', 'vendredi', 'samedi'],
95                 'creneaux' => $creneaux,
96                 'grille' => $grille,
97                 'totalHeures' => round($totalMinutes / 60, 1),
98                 'annee' => AnneeScolaire::enCours(),
99             ]);
100        }

107        public function store(Request $request): RedirectResponse
108        {
109            $data = $request->validate([
110                'nom' => ['required', 'string', 'max:50'],
111                'postnom' => ['required', 'string', 'max:50'],
112                'prenom' => ['required', 'string', 'max:50'],
113                'sexe' => ['required', 'in:M,F'],
114                'specialite' => ['required', 'string', 'max:100'],
115                'telephone' => ['required', 'string', 'max:20'],
116                'email' => ['required', 'email', 'max:100', 'unique:professeurs,email'],
117                'photo' => ['nullable', 'image', 'mimes:jpg,jpeg,png', 'max:2048'],
118                'statut' => ['required', 'in:actif,inactif'],
119            ], [
120                'email.unique' => 'Cette adresse e-mail est déjà utilisée.',
121                'photo.max' => 'La photo ne doit pas dépasser 2 Mo.',
122                'photo.mimes' => 'La photo doit être au format JPG, JPEG ou PNG.',
123            ]);
124    
125            $year = now()->format('Y');
126            $number = Professeurs::where('matricule', 'like', 'PROF-'.$year.'-%')->count() + 1;
127            do {
128                $data['matricule'] = 'PROF-'.$year.'-'.str_pad((string) $number, 4, '0', STR_PAD_LEFT);
129                $number++;
130            } while (Professeurs::where('matricule', $data['matricule'])->exists());
131    
132            if ($request->hasFile('photo')) {
133                $data['photo'] = $request->file('photo')->store('professeurs/photos', 'public');
134            }
135    
136            Professeurs::create($data);
137    
138            return redirect()->route('admin.professeurs.create')->with('success', 'Le professeur a été créé avec succès.');
139        }
