509    test('une réaction émoji est enregistrée, affichée et retirée', function () {
510        $alice = User::factory()->create(['role' => 'admin']);
511        $bruno = User::factory()->create(['role' => 'professeur']);
512        $conversation = Messagerie::creer($alice, [$bruno->id], 'Sujet', 'Coucou');
513        $message = $conversation->messages()->first();
514    
515        $this->actingAs($bruno)
516            ->postJson(route('messagerie.reagir', $message), ['reaction' => '👍'])
517            ->assertOk()
518            ->assertJsonPath('message.id', $message->id);
519    
520        expect($message->fresh()->reaction)->toBe('👍')
521            ->and($message->fresh()->aReaction())->toBeTrue();
522    
523        // Un non-participant ne peut pas réagir.
524        $etranger = User::factory()->create(['role' => 'comptable']);
525        $this->actingAs($etranger)->postJson(route('messagerie.reagir', $message), ['reaction' => '❤️'])->assertForbidden();
526    
527        // Réaction retirée.
528        $this->actingAs($bruno)->postJson(route('messagerie.reagir', $message), ['reaction' => ''])->assertOk();
529        expect($message->fresh()->reaction)->toBeNull();
530    });
