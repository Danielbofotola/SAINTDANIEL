13         protected function casts(): array
14         {
15             return [
16                 'date_evaluation' => 'date',
17                 'max_note' => 'decimal:2',
18             ];
19         }
20     
21         public function affectation()
22         {
23             return $this->belongsTo(AffectationCours::class, 'affectation_id');

26         public function periode()
27         {
28             return $this->belongsTo(Periode::class);

31         public function notes()
