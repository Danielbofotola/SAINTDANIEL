120        reglerPortailParents(true);

122        $parent = User::factory()->create(['role' => 'parent', 'nom' => 'Bofotola', 'prenom' => 'Daniel']);

187    test('un message vocal est enregistré et rendu par un lecteur audio', function () {
188        Storage::fake('public');
189        reglerPortailParents(true);
190    
191        $parent = User::factory()->create(['role' => 'parent']);
192        $prof = User::factory()->create(['role' => 'professeur']);
193        $conversation = Messagerie::creer($prof, [$parent->id], 'Vocal', 'Envoyez-moi un vocal.');
194    
195        $this->actingAs($parent)
196            ->post(route('messagerie.repondre', $conversation), [
197                'piece_jointe' => fichierAudioDeTest('vocal.wav'),
198            ])
199            ->assertSessionHasNoErrors()
200            ->assertRedirect();
201    
202        $message = $conversation->messages()->latest('id')->first();
203    
204        expect($message->piece_jointe)->not->toBeNull()
205            ->and($message->estAudio())->toBeTrue();
206    
207        $html = view('messagerie._bulle', ['message' => $message, 'user' => $parent, 'active' => $conversation])->render();
208    
209        expect($html)->toContain('bulle__audio')->toContain('<audio');
210    });

226    /**
227     * Petit fichier audio WAVE valide, pour tester les pièces jointes vocales sans
228     * dépendre d'un encodeur : la règle « mimes » détecte le type par le contenu.
229     */
230    function fichierAudioDeTest(string $nom = 'vocal.wav'): \Illuminate\Http\UploadedFile
231    {
232        $entete = 'RIFF'.pack('V', 36).'WAVEfmt '.pack('V', 16).pack('v', 1).pack('v', 1)
233            .pack('V', 8000).pack('V', 8000).pack('v', 1).pack('v', 8).'data'.pack('V', 0);
234    
235        $chemin = tempnam(sys_get_temp_dir(), 'wmark_').'.wav';
236        file_put_contents($chemin, $entete);
237    
238        return new \Illuminate\Http\UploadedFile($chemin, $nom, 'audio/wav', null, true);
239    }
