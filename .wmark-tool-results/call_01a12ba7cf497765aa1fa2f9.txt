120        public function updateAccount(Request $request, User $user): RedirectResponse
121        {
122            if ($user->is($request->user())) {
123                return redirect()->route('admin.profile.edit');
124            }
125    
126            $data = $request->validate([
127                'nom' => ['nullable', 'string', 'max:50'],
128                'postnom' => ['nullable', 'string', 'max:50'],
129                'prenom' => ['nullable', 'string', 'max:50'],
130                'date_naissance' => ['nullable', 'date', 'before:today'],
131                'username' => ['required', 'string', 'max:50', Rule::unique('users', 'username')->ignore($user->id)],
132                'email' => ['required', 'email', 'max:100', Rule::unique('users', 'email')->ignore($user->id)],
133                'telephone' => ['nullable', 'string', 'max:20'],
134                'adresse' => ['nullable', 'string', 'max:500'],
135                'photo' => ['nullable', 'image', 'mimes:jpg,jpeg,png', 'max:2048'],
136                'role' => ['required', Rule::in(User::ROLES)],
137                'statut' => ['required', 'in:actif,inactif'],
138                'professeur_id' => [
139                    'nullable',
140                    'exists:professeurs,id',
141                    Rule::unique('users', 'professeur_id')->ignore($user->id),
142                    Rule::requiredIf(fn () => $request->input('role') === 'professeur'),
143                ],
144                'enfants' => ['nullable', 'array'],
145                'enfants.*' => ['integer', 'exists:eleves,id'],
146            ]);
147    
148            if ($request->hasFile('photo')) {
149                if ($user->photo) {
150                    Storage::disk('public')->delete($user->photo);
151                }
152    
153                $data['photo'] = $request->file('photo')->store('users/photos', 'public');
154            }
155    
156            $etaitParent = $user->role === 'parent';
157    
158            $user->update($data);
159    
160            // Seul un compte parent porte des enfants ; un ancien compte parent en est
161            // détaché lorsqu'il change de rôle.
162            if ($data['role'] === 'parent') {
163                $user->eleves()->sync($data['enfants'] ?? []);
164            } elseif ($etaitParent) {
165                $user->eleves()->sync([]);
166            }
167    
168            return redirect()->route('admin.profile.accounts')->with('success', 'Le profil utilisateur a été mis à jour.');
169        }

184        /**
185         * Réinitialise le mot de passe (et le code PIN) d'un autre compte.
186         */
187        public function updateAccountPassword(Request $request, User $user): RedirectResponse
188        {
189            if ($user->is($request->user())) {
190                return redirect()->route('admin.profile.edit');
191            }
192    
193            $data = $request->validate([
194                'password' => ['required', 'string', 'min:8', 'confirmed'],
195                'pin_code' => ['nullable', 'digits:4'],
196                'remove_pin' => ['nullable', 'boolean'],
197            ], [
198                'password.required' => 'Le nouveau mot de passe est obligatoire.',
199                'password.min' => 'Le mot de passe doit contenir au moins 8 caractères.',
200                'password.confirmed' => 'La confirmation du mot de passe ne correspond pas.',
201                'pin_code.digits' => 'Le code PIN doit contenir exactement 4 chiffres.',
202            ]);
203    
204            unset($data['remove_pin']);
205    
206            // Le cast « hashed » du modèle hache le mot de passe.
207            if (filled($data['pin_code'] ?? null)) {
208                $data['pin_code'] = Hash::make($data['pin_code']);
209            } elseif ($request->boolean('remove_pin')) {
210                $data['pin_code'] = null;
211            } else {
212                unset($data['pin_code']);
213            }
214    
215            $user->update($data);
216    
217            Log::notice('Admin reset an account password', [
218                'admin_id' => $request->user()->id,
219                'user_id' => $user->id,
220                'pin_changed' => array_key_exists('pin_code', $data),
221            ]);
222    
223            return redirect()
224                ->route('admin.profile.accounts')
225                ->with('success', "Les identifiants de {$user->username} ont été réinitialisés.");
226        }

228        public function impersonate(Request $request, User $user): RedirectResponse
229        {
230            if ($request->session()->has('impersonating_admin_id')) {
231                return back()->with('error', 'Une session de consultation est déjà en cours.');
232            }
233    
234            if ($user->is($request->user()) || $user->role === 'admin' || $user->statut !== 'actif') {
235                return back()->with('error', 'Seuls les autres comptes actifs non administrateurs peuvent être ouverts.');
236            }
237    
238            $admin = $request->user();
239            $request->session()->put('impersonating_admin_id', $admin->id);
240            auth()->login($user);
241            $request->session()->regenerate();
242    
243            Log::notice('Admin started account impersonation', [
244                'admin_id' => $admin->id,
245                'user_id' => $user->id,
246            ]);
247    
248            return redirect()->route($user->dashboardRouteName());
249        }

251        public function stopImpersonation(Request $request): RedirectResponse
252        {
253            $admin = User::query()
254                ->whereKey($request->session()->get('impersonating_admin_id'))
255                ->where('role', 'admin')
256                ->where('statut', 'actif')
257                ->first();
258    
259            if (! $admin) {
260                auth()->logout();
261                $request->session()->invalidate();
262                $request->session()->regenerateToken();
263    
264                return redirect()->route('login')->with('error', 'La session administrateur d’origine n’est plus disponible.');
265            }
266    
267            Log::notice('Admin stopped account impersonation', [
268                'admin_id' => $admin->id,
269                'user_id' => $request->user()->id,
270            ]);
271    
272            auth()->login($admin);
273            $request->session()->forget('impersonating_admin_id');
274            $request->session()->regenerate();
275    
276            return redirect()->route('admin.dashboard')->with('success', 'Vous êtes de retour sur votre compte administrateur.');
277        }
