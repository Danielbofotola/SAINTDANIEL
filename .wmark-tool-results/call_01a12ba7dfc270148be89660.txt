371                'email.unique' => "Cette adresse e-mail est déjà utilisée.",
372            ]);
373    
374            if ($request->hasFile('photo')) {
375                if ($user->photo) {
376                    Storage::disk('public')->delete($user->photo);
377                }
378    
379                $data['photo'] = $request->file('photo')->store('users/photos', 'public');
380            }
381    
382            $user->update($data);
383    
384            return redirect()->route('profile.show')->with('success', 'Les propriétés du profil ont été modifiées.');
385        }

417        public function toggleAccount(User $user): RedirectResponse
418        {
419            if ($user->is(auth()->user())) {
420                return back()->with('error', 'Vous ne pouvez pas désactiver votre propre compte depuis cette page.');
421            }
422    
423            $user->update(['statut' => $user->statut === 'actif' ? 'inactif' : 'actif']);
424    
425            if ($user->statut !== 'actif') {
426                $this->purgerSessions($user);
427            }
428    
429            return back()->with('success', 'Le statut du compte a été mis à jour.');
430        }

432        public function destroyAccount(User $user): RedirectResponse
433        {
434            if ($user->is(auth()->user())) {
435                return back()->with('error', 'Vous ne pouvez pas supprimer votre propre compte.');
436            }
437    
438            $this->purgerSessions($user);
439            $user->delete();
440    
441            return back()->with('success', 'Le compte a été supprimé définitivement.');
442        }

444        /**
445         * Met fin aux sessions actives d'un compte (driver « database ») : un
446         * utilisateur désactivé ou supprimé ne reste pas connecté.
447         */
448        private function purgerSessions(User $user): void
449        {
450            if (config('session.driver') === 'database' && Schema::hasTable('sessions')) {
451                DB::table('sessions')->where('user_id', $user->id)->delete();
452            }
453        }
