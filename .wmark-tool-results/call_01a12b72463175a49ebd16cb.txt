1      <?php
2      
3      namespace App\Support;
4      
5      use App\Models\Eleve;
6      use App\Models\FraisScolaire;
7      use App\Models\Inscription;
8      use App\Models\Paiement;
9      use Illuminate\Support\Collection;
10     
11     /**
12      * Situation complète d'un élève : parcours année par année (années passées
13      * comprises) et état financier — frais exigés, paiements encaissés, dette.
14      *
15      * Une seule logique, partagée par le dossier de l'administration et celui du
16      * préfet des études, pour que les deux affichent exactement les mêmes chiffres.
17      */
18     class SituationEleve
19     {
20         /**
21          * État financier d'une inscription : exigible, encaissé, reste à payer.
22          *
23          * @return array{attendu: float, paye: float, reste: float}
24          */
25         public static function finances(Inscription $inscription): array
26         {
27             $attendu = (float) FraisScolaire::query()
28                 ->where('classe_id', $inscription->classe_id)
29                 ->where('annee_scolaire_id', $inscription->annee_scolaire_id)
30                 ->where('statut', 'actif')
31                 ->sum('montant_exige');
32     
33             $paye = (float) Paiement::query()
34                 ->where('inscription_id', $inscription->id)
35                 ->where('statut', 'actif')
36                 ->sum('montant_paye');
37     
38             return ['attendu' => $attendu, 'paye' => $paye, 'reste' => round($attendu - $paye, 2)];
39         }
40     
41         /**
42          * Parcours de l'élève : toutes ses inscriptions, de l'année la plus récente
43          * à la plus ancienne, enrichies de l'état financier de chaque année.
44          *
45          * @return Collection<int, array{inscription: Inscription, finances: array{attendu: float, paye: float, reste: float}, frais: array}>
46          */
47         public static function parcours(Eleve $eleve): Collection
48         {
49             return $eleve->inscriptions()
50                 ->with(['classe.cycle', 'classe.section', 'classe.option', 'anneeScolaire', 'bulletins'])
51                 ->get()
52                 ->sortByDesc(fn (Inscription $inscription) => (string) ($inscription->anneeScolaire?->libelle ?? ''))
53                 ->values()
54                 ->map(fn (Inscription $inscription) => [
55                     'inscription' => $inscription,
56                     'finances' => self::finances($inscription),
57                     'frais' => self::frais($inscription),
58                 ]);
59         }
60     
61         /**
62          * Détail des frais exigés d'une inscription : chaque ligne de frais, ce qui a
63          * été encaissé dessus et le reste à payer. Les paiements non rattachés à une
64          * ligne sont comptés à part, pour que le détail rejoigne le total de l'année.
65          *
66          * @return array{lignes: Collection<int, array{frais: FraisScolaire, exige: float, paye: float, reste: float}>, non_affectes: float, totaux: array{exige: float, paye: float, reste: float}}
67          */
68         public static function frais(Inscription $inscription): array
69         {
70             $lignes = FraisScolaire::query()
71                 ->with('typeFrais')
72                 ->where('classe_id', $inscription->classe_id)
73                 ->where('annee_scolaire_id', $inscription->annee_scolaire_id)
74                 ->where('statut', 'actif')
75                 ->orderBy('date_echeance')
76                 ->orderBy('id')
77                 ->get();
78     
79             $paiements = Paiement::query()
80                 ->where('inscription_id', $inscription->id)
81                 ->where('statut', 'actif')
82                 ->get(['frais_scolaire_id', 'montant_paye']);
83     
84             // Montant encaissé par ligne de frais.
85             $parFrais = [];
86             foreach ($paiements as $paiement) {
87                 if ($paiement->frais_scolaire_id === null) {
88                     continue;
89                 }
90                 $cle = (int) $paiement->frais_scolaire_id;
91                 $parFrais[$cle] = ($parFrais[$cle] ?? 0.0) + (float) $paiement->montant_paye;
92             }
93     
94             $exigeTotal = round((float) $lignes->sum('montant_exige'), 2);
95             $payeTotal = round((float) $paiements->sum('montant_paye'), 2);
96     
97             $detail = $lignes->map(function (FraisScolaire $frais) use ($parFrais) {
98                 $exige = (float) $frais->montant_exige;
99                 $paye = round($parFrais[(int) $frais->id] ?? 0.0, 2);
100    
101                return ['frais' => $frais, 'exige' => $exige, 'paye' => $paye, 'reste' => round($exige - $paye, 2)];
102            });
103    
104            return [
105                'lignes' => $detail,
106                'non_affectes' => round($payeTotal - (float) $detail->sum('paye'), 2),
107                'totaux' => [
108                    'exige' => $exigeTotal,
109                    'paye' => $payeTotal,
110                    'reste' => round($exigeTotal - $payeTotal, 2),
111                ],
112            ];
113        }
114    
115        /**
116         * Détail des paiements d'une inscription, du plus récent au plus ancien.
117         *
118         * @return Collection<int, Paiement>
119         */
120        public static function paiements(Inscription $inscription): Collection
121        {
122            return Paiement::query()
123                ->with('fraisScolaire.typeFrais')
124                ->where('inscription_id', $inscription->id)
125                ->orderByDesc('date_paiement')
126                ->orderByDesc('id')
127                ->get();
128        }
129    
130        /**
131         * Dette cumulée sur l'ensemble du parcours.
132         *
133         * @param  Collection<int, array{inscription: Inscription, finances: array{attendu: float, paye: float, reste: float}}>  $parcours
134         */
135        public static function detteTotale(Collection $parcours): float
136        {
137            return round($parcours->sum(fn (array $ligne) => $ligne['finances']['reste']), 2);
138        }
139    }
