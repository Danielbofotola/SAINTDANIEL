1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use App\Models\Classe;
7      use App\Models\Cycle;
8      use App\Models\Eleve;
9      use App\Models\Option;
10     use App\Models\Professeurs;
11     use App\Models\Section;
12     use App\Models\User;
13     use Illuminate\Http\Request;
14     use Illuminate\View\View;
15     
16     class PrefectureDashboardController extends Controller
17     {
18         /**
19          * Espace préfecture : vue de supervision de l'établissement pour l'année en cours.
20          *
21          * Consultation seule — aucun écran d'administration n'est exposé ici.
22          */
23         public function index(Request $request): View
24         {
25             return view('prefecture.dashboard', [
26                 'user' => $request->user(),
27                 'schoolYear' => AnneeScolaire::enCours(),
28                 'effectifTotal' => Eleve::inscritAnneeEnCours()->where('statut', 'actif')->count(),
29                 'classesActives' => Classe::where('statut', 'actif')->count(),
30                 'professeursActifs' => Professeurs::where('statut', 'actif')->count(),
31                 'sectionsCount' => Section::count(),
32                 'optionsCount' => Option::count(),
33                 'effectifsParCycle' => Cycle::query()
34                     ->where('statut', 'actif')
35                     ->withCount(['inscriptions as effectif' => fn ($query) => $query
36                         ->anneeEnCours()
37                         ->where('inscriptions.statut', 'actif')])
38                     ->orderBy('nom')
39                     ->get(),
40                 'classes' => Classe::query()
41                     ->with(['cycle', 'salle'])
42                     ->withCount(['inscriptions as effectif' => fn ($query) => $query
43                         ->anneeEnCours()
44                         ->where('statut', 'actif')])
45                     ->where('statut', 'actif')
46                     ->orderBy('nom')
47                     ->get(),
48                 'personnel' => User::query()
49                     ->selectRaw('role, COUNT(*) as total')
50                     ->groupBy('role')
51                     ->pluck('total', 'role'),
52             ]);
53         }
54     
55         /**
56          * Effectifs : répartition par cycle, par classe et par sexe (année en cours).
57          */
58         public function effectifs(Request $request): View
59         {
60             $parSexe = Eleve::query()
61                 ->inscritAnneeEnCours()
62                 ->where('statut', 'actif')
63                 ->selectRaw('sexe, COUNT(*) as total')
64                 ->groupBy('sexe')
65                 ->pluck('total', 'sexe');
66     
67             $classes = Classe::query()
68                 ->with('cycle')
69                 ->withCount(['inscriptions as effectif' => fn ($query) => $query
70                     ->anneeEnCours()
71                     ->where('inscriptions.statut', 'actif')])
72                 ->where('statut', 'actif')
73                 ->orderBy('nom')
74                 ->get();
75     
76             return view('prefecture.effectifs', [
77                 'user' => $request->user(),
78                 'schoolYear' => AnneeScolaire::enCours(),
79                 'effectifTotal' => (int) $parSexe->sum(),
80                 'parSexe' => $parSexe,
81                 'cycles' => Cycle::query()
82                     ->where('statut', 'actif')
83                     ->withCount(['inscriptions as effectif' => fn ($query) => $query
84                         ->anneeEnCours()
85                         ->where('inscriptions.statut', 'actif')])
86                     ->orderBy('nom')
87                     ->get(),
88                 'classesParCycle' => $classes->groupBy(fn (Classe $classe) => $classe->cycle?->nom ?? 'Sans cycle'),
89                 'nbClasses' => $classes->count(),
90             ]);
91         }
92     
93         /**
94          * Classes actives : organisation, locaux et effectifs.
95          */
96         public function classes(Request $request): View
97         {
98             return view('prefecture.classes', [
99                 'user' => $request->user(),
100                'schoolYear' => AnneeScolaire::enCours(),
101                'classes' => Classe::query()
102                    ->with(['cycle', 'salle', 'section', 'option'])
103                    ->withCount(['inscriptions as effectif' => fn ($query) => $query
104                        ->anneeEnCours()
105                        ->where('inscriptions.statut', 'actif')])
106                    ->where('statut', 'actif')
107                    ->orderBy('nom')
108                    ->get(),
109            ]);
110        }
111    
112        /**
113         * Personnel : comptes utilisateurs par rôle et enseignants enregistrés.
114         */
115        public function personnel(Request $request): View
116        {
117            return view('prefecture.personnel', [
118                'user' => $request->user(),
119                'schoolYear' => AnneeScolaire::enCours(),
120                'parRole' => User::query()
121                    ->selectRaw('role, COUNT(*) as total')
122                    ->groupBy('role')
123                    ->pluck('total', 'role'),
124                'comptes' => User::query()
125                    ->orderBy('role')
126                    ->orderBy('nom')
127                    ->orderBy('prenom')
128                    ->get(),
129                'professeurs' => Professeurs::query()->orderBy('nom')->orderBy('prenom')->get(),
130            ]);
131        }
132    
133        /**
134         * Profil du préfet : identité, coordonnées, activité et périmètre du rôle.
135         */
136        public function profil(Request $request): View
137        {
138            $utilisateur = $request->user();
139    
140            return view('prefecture.profil', [
141                'user' => $utilisateur,
142                'schoolYear' => AnneeScolaire::enCours(),
143                'notificationsTotal' => $utilisateur->notifications()->count(),
144                'notificationsNonLues' => $utilisateur->notifications()->whereNull('read_at')->count(),
145                'dernieresNotifications' => $utilisateur->notifications()->latest()->limit(5)->get(),
146                'effectifTotal' => Eleve::inscritAnneeEnCours()->where('statut', 'actif')->count(),
147                'classesActives' => Classe::where('statut', 'actif')->count(),
148                'professeursActifs' => Professeurs::where('statut', 'actif')->count(),
149                'comptesTotal' => User::count(),
150            ]);
151        }
152    }
