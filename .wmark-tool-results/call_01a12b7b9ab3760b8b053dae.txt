1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use Illuminate\Http\RedirectResponse;
7      use Illuminate\Http\Request;
8      use Illuminate\Support\Facades\DB;
9      use Illuminate\Validation\Rule;
10     use Illuminate\View\View;
11     
12     class AnneeScolaireController extends Controller
13     {
14         /**
15          * Affiche le formulaire de création et la liste des années scolaires.
16          */
17         public function index(): View
18         {
19             return view('admin.annees.index', [
20                 'annees' => AnneeScolaire::orderByDesc('date_debut')
21                     ->orderByDesc('libelle')
22                     ->get(),
23             ]);
24         }
25     
26         /**
27          * Enregistre une nouvelle année scolaire.
28          */
29         public function store(Request $request): RedirectResponse
30         {
31             $data = $this->validateData($request);
32     
33             $data['statut'] = 'actif';
34             $data['is_encours'] = false;
35     
36             AnneeScolaire::create($data);
37     
38             return redirect()
39                 ->route('admin.annees.index')
40                 ->with('success', "L'année scolaire « {$data['libelle']} » a été créée avec succès.");
41         }
42     
43         /**
44          * Formulaire de modification d'une année scolaire.
45          */
46         public function edit(AnneeScolaire $annee): View
47         {
48             return view('admin.annees.edit', compact('annee'));
49         }
50     
51         /**
52          * Met à jour une année scolaire existante.
53          */
54         public function update(Request $request, AnneeScolaire $annee): RedirectResponse
55         {
56             $annee->update($this->validateData($request, $annee));
57     
58             return redirect()
59                 ->route('admin.annees.index')
60                 ->with('success', "L'année scolaire « {$annee->libelle} » a été modifiée avec succès.");
61         }
62     
63         /**
64          * Supprime une année scolaire, sauf si des données y sont rattachées.
65          */
66         public function destroy(AnneeScolaire $annee): RedirectResponse
67         {
68             if ($annee->isUtilisee()) {
69                 return back()->with('error', "Suppression impossible : l'année « {$annee->libelle} » est déjà rattachée à des inscriptions, périodes ou données financières.");
70             }
71     
72             $libelle = $annee->libelle;
73             $annee->delete();
74     
75             return redirect()
76                 ->route('admin.annees.index')
77                 ->with('success', "L'année scolaire « {$libelle} » a été supprimée.");
78         }
79     
80         /**
81          * Active / désactive une année scolaire.
82          *
83          * Une seule année peut être « en cours » à la fois : activer une année
84          * désactive automatiquement les autres.
85          */
86         public function toggle(AnneeScolaire $annee): RedirectResponse
87         {
88             $activation = ! $annee->is_encours;
89     
90             DB::transaction(function () use ($annee, $activation) {
91                 AnneeScolaire::query()->update(['is_encours' => false]);
92                 // Mise à jour par requête : `$annee->update()` serait ignoré
93                 // si sa valeur en mémoire vaut déjà $activation (dirty tracking).
94                 AnneeScolaire::whereKey($annee->id)->update(['is_encours' => $activation]);
95             });
96     
97             return back()->with(
98                 'success',
99                 $activation
100                    ? "L'année « {$annee->libelle} » est désormais l'année en cours."
101                    : "L'année « {$annee->libelle} » n'est plus l'année en cours."
102            );
103        }
104    
105        /**
106         * Règles de validation communes à la création et à la modification.
107         */
108        private function validateData(Request $request, ?AnneeScolaire $annee = null): array
109        {
110            return $request->validate([
111                'libelle' => ['required', 'string', 'max:20', Rule::unique('annees_scolaires', 'libelle')->ignore($annee?->id)],
112                'date_debut' => ['required', 'date'],
113                'date_fin' => ['required', 'date', 'after:date_debut'],
114            ]);
115        }
116    }
