1      <?php
2      
3      namespace App\Models;
4      
5      // use Illuminate\Contracts\Auth\MustVerifyEmail;
6      use Database\Factories\UserFactory;
7      use Illuminate\Database\Eloquent\Factories\HasFactory;
8      use Illuminate\Foundation\Auth\User as Authenticatable;
9      use Illuminate\Notifications\Notifiable;
10     
11     class User extends Authenticatable
12     {
13         /** @use HasFactory<UserFactory> */
14         use HasFactory, Notifiable;
15     
16         public function professeur()
17         {
18             return $this->belongsTo(Professeurs::class, 'professeur_id');
19         }
20     
21         /**
22          * Nom de route du tableau de bord correspondant au rôle.
23          *
24          * Source unique : login, « retour au tableau de bord » et impersonation s'y réfèrent.
25          * Le repli est `profile.show` — jamais `home`, qui redirige selon le rôle (boucle).
26          */
27         public static function dashboardRouteNameFor(?string $role): string
28         {
29             return match ($role) {
30                 'admin' => 'admin.dashboard',
31                 'prefetture' => 'prefecture.dashboard',
32                 'titulaire' => 'titulaire.dashboard',
33                 'professeur' => 'professeur.dashboard',
34                 'comptable' => 'comptable.dashboard',
35                 default => 'profile.show',
36             };
37         }
38     
39         public function dashboardRouteName(): string
40         {
41             return static::dashboardRouteNameFor($this->role);
42         }
43     
44         /**
45          * The attributes that are mass assignable.
46          *
47          * @var list<string>
48          */
49         protected $fillable = [
50             'professeur_id',
51             'nom',
52             'postnom',
53             'prenom',
54             'date_naissance',
55             'username',
56             'email',
57             'telephone',
58             'adresse',
59             'photo',
60             'password',
61             'pin_code',
62             'role',
63             'statut',
64             'last_login',
65         ];
66     
67         /**
68          * The attributes that should be hidden for serialization.
69          *
70          * @var list<string>
71          */
72         protected $hidden = [
73             'password',
74             'pin_code',
75             'remember_token',
76         ];
77     
78         /**
79          * Get the attributes that should be cast.
80          *
81          * @return array<string, string>
82          */
83         protected function casts(): array
84         {
85             return [
86                 'email_verified_at' => 'datetime',
87                 'password' => 'hashed',
88                 'last_login' => 'datetime',
89                 'date_naissance' => 'date',
90             ];
91         }
92     }
