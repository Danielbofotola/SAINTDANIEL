27         public function storeAccount(Request $request): RedirectResponse
28         {
29             $data = $request->validate([
30                 'nom' => ['nullable', 'string', 'max:50'],
31                 'postnom' => ['nullable', 'string', 'max:50'],
32                 'prenom' => ['nullable', 'string', 'max:50'],
33                 'username' => ['required', 'string', 'max:50', 'unique:users,username'],
34                 'email' => ['required', 'email', 'max:100', 'unique:users,email'],
35                 'telephone' => ['nullable', 'string', 'max:20'],
36                 'professeur_id' => [
37                     'nullable',
38                     'exists:professeurs,id',
39                     Rule::unique('users', 'professeur_id'),
40                     Rule::requiredIf(fn () => $request->input('role') === 'professeur'),
41                 ],
42                 'role' => ['required', 'in:admin,prefetture,titulaire,professeur,comptable'],
43                 'statut' => ['required', 'in:actif,inactif'],
44                 'password' => ['required', 'string', 'min:8', 'confirmed'],
45                 'pin_code' => ['nullable', 'digits:4'],
46             ], [
47                 'username.unique' => "Ce nom d'utilisateur est déjà utilisé.",
48                 'email.unique' => "Cette adresse e-mail est déjà utilisée.",
49                 'password.min' => 'Le mot de passe doit contenir au moins 8 caractères.',
50                 'password.confirmed' => 'La confirmation du mot de passe ne correspond pas.',
51                 'pin_code.digits' => 'Le code PIN doit contenir exactement 4 chiffres.',
52                 'professeur_id.required' => 'Un professeur doit être lié à un compte ayant le rôle professeur.',
53                 'professeur_id.unique' => 'Ce professeur est déjà lié à un autre compte utilisateur.',
54             ]);
55     
56             $data['pin_code'] = filled($data['pin_code'] ?? null) ? Hash::make($data['pin_code']) : null;
57     
58             User::create($data);
59     
60             return redirect()->route('admin.profile.accounts')->with('success', 'Le compte utilisateur a été créé.');
61         }

91         public function updateAccount(Request $request, User $user): RedirectResponse
92         {
93             if ($user->is($request->user())) {
94                 return redirect()->route('admin.profile.edit');
95             }
96     
97             $data = $request->validate([
98                 'nom' => ['nullable', 'string', 'max:50'],
99                 'postnom' => ['nullable', 'string', 'max:50'],
100                'prenom' => ['nullable', 'string', 'max:50'],
101                'date_naissance' => ['nullable', 'date', 'before:today'],
102                'username' => ['required', 'string', 'max:50', Rule::unique('users', 'username')->ignore($user->id)],
103                'email' => ['required', 'email', 'max:100', Rule::unique('users', 'email')->ignore($user->id)],
104                'telephone' => ['nullable', 'string', 'max:20'],
105                'adresse' => ['nullable', 'string', 'max:500'],
106                'photo' => ['nullable', 'image', 'mimes:jpg,jpeg,png', 'max:2048'],
107                'role' => ['required', 'in:admin,prefetture,titulaire,professeur,comptable'],
108                'statut' => ['required', 'in:actif,inactif'],
109                'professeur_id' => [
110                    'nullable',
111                    'exists:professeurs,id',
112                    Rule::unique('users', 'professeur_id')->ignore($user->id),
113                    Rule::requiredIf(fn () => $request->input('role') === 'professeur'),
114                ],
115            ]);
116    
117            if ($request->hasFile('photo')) {
118                if ($user->photo) {
119                    Storage::disk('public')->delete($user->photo);
120                }
121    
122                $data['photo'] = $request->file('photo')->store('users/photos', 'public');
123            }
124    
125            $user->update($data);
126    
127            return redirect()->route('admin.profile.accounts')->with('success', 'Le profil utilisateur a été mis à jour.');
128        }

130        public function impersonate(Request $request, User $user): RedirectResponse
131        {
132            if ($request->session()->has('impersonating_admin_id')) {
133                return back()->with('error', 'Une session de consultation est déjà en cours.');
134            }
135    
136            if ($user->is($request->user()) || $user->role === 'admin' || $user->statut !== 'actif') {
137                return back()->with('error', 'Seuls les autres comptes actifs non administrateurs peuvent être ouverts.');
138            }
139    
140            $admin = $request->user();
141            $request->session()->put('impersonating_admin_id', $admin->id);
142            auth()->login($user);
143            $request->session()->regenerate();
144    
145            Log::notice('Admin started account impersonation', [
146                'admin_id' => $admin->id,
147                'user_id' => $user->id,
148            ]);
149    
150            $dashboard = match ($user->role) {
151                'prefetture' => 'prefecture.dashboard',
152                'titulaire' => 'titulaire.dashboard',
153                'professeur' => 'professeur.dashboard',
154                'comptable' => 'comptable.dashboard',
155                default => 'home',
156            };
157    
158            return redirect()->route($dashboard);
159        }

161        public function stopImpersonation(Request $request): RedirectResponse
162        {
163            $admin = User::query()
164                ->whereKey($request->session()->get('impersonating_admin_id'))
165                ->where('role', 'admin')
166                ->where('statut', 'actif')
167                ->first();
168    
169            if (! $admin) {
170                auth()->logout();
171                $request->session()->invalidate();
172                $request->session()->regenerateToken();
173    
174                return redirect()->route('login')->with('error', 'La session administrateur d’origine n’est plus disponible.');
175            }
176    
177            Log::notice('Admin stopped account impersonation', [
178                'admin_id' => $admin->id,
179                'user_id' => $request->user()->id,
180            ]);
181    
182            auth()->login($admin);
183            $request->session()->forget('impersonating_admin_id');
184            $request->session()->regenerate();
185    
186            return redirect()->route('admin.dashboard')->with('success', 'Vous êtes de retour sur votre compte administrateur.');
187        }
