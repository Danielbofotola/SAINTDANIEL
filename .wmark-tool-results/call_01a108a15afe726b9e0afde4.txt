public function show(Request $request): View
    {
        return view('admin.profile.show', [
            'user' => $request->user(),
            'users' => User::with('professeur')
                ->orderByRaw("CASE WHEN role = 'admin' THEN 0 ELSE 1 END")
                ->orderBy('username')
                ->limit(5)
                ->get(),
        ]);
    }
