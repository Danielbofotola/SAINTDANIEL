1      <?php
2      
3      namespace App\Http\Controllers;
4      
5      use App\Models\AnneeScolaire;
6      use App\Models\CategorieDepense;
7      use App\Models\Depense;
8      use Illuminate\Http\RedirectResponse;
9      use Illuminate\Http\Request;
10     use Illuminate\Validation\Rule;
11     use Illuminate\View\View;
12     
13     class DepenseController extends Controller
14     {
15         public function index(Request $request): View
16         {
17             $annee = AnneeScolaire::enCours();
18     
19             $depenses = Depense::query()
20                 ->anneeEnCours()
21                 ->with('categorie')
22                 ->latest('date_depense')
23                 ->latest('id')
24                 ->paginate(25)
25                 ->withQueryString();
26     
27             return view('admin.depenses.index', [
28                 'depenses' => $depenses,
29                 'annee' => $annee,
30                 'categories' => CategorieDepense::query()->withCount('depenses')->orderBy('nom')->get(),
31                 'categoriesActives' => CategorieDepense::where('statut', 'actif')->orderBy('nom')->get(),
32                 'totalDepenses' => $annee
33                     ? (float) Depense::query()->where('annee_scolaire_id', $annee->id)->where('statut', 'actif')->sum('montant')
34                     : 0.0,
35             ]);
36         }
37     
38         public function store(Request $request): RedirectResponse
39         {
40             $annee = AnneeScolaire::enCours();
41     
42             if (! $annee) {
43                 return back()->withInput()->with('error', 'Aucune année scolaire en cours : impossible d’enregistrer une dépense.');
44             }
45     
46             $data = $request->validate([
47                 'category_id' => ['required', Rule::exists('categories_depenses', 'id')->where('statut', 'actif')],
48                 'motif' => ['required', 'string', 'max:255'],
49                 'montant' => ['required', 'numeric', 'min:0.01'],
50                 'date_depense' => ['required', 'date'],
51                 'piece_justificative' => ['nullable', 'string', 'max:255'],
52             ], [
53                 'category_id.required' => 'Sélectionnez une catégorie.',
54                 'motif.required' => 'Le motif de la dépense est obligatoire.',
55                 'montant.required' => 'Le montant est obligatoire.',
56                 'montant.min' => 'Le montant doit être supérieur à zéro.',
57                 'date_depense.required' => 'La date de la dépense est obligatoire.',
58             ]);
59     
60             Depense::create($data + ['annee_scolaire_id' => $annee->id, 'statut' => 'actif']);
61     
62             return redirect()->route('admin.depenses.index')->with('success', 'La dépense a été enregistrée.');
63         }
64     
65         public function destroy(Depense $depense): RedirectResponse
66         {
67             $depense->delete();
68     
69             return redirect()->route('admin.depenses.index')->with('success', 'La dépense a été supprimée.');
70         }
71     
72         public function storeCategorie(Request $request): RedirectResponse
73         {
74             $data = $request->validate([
75                 'nom' => ['required', 'string', 'max:100', 'unique:categories_depenses,nom'],
76             ], [
77                 'nom.required' => 'Le nom de la catégorie est obligatoire.',
78                 'nom.unique' => 'Cette catégorie existe déjà.',
79             ]);
80     
81             CategorieDepense::create($data + ['statut' => 'actif']);
82     
83             return redirect()->route('admin.depenses.index')->with('success', 'La catégorie a été créée.');
84         }
85     
86         public function destroyCategorie(CategorieDepense $categorieDepense): RedirectResponse
87         {
88             if ($categorieDepense->depenses()->exists()) {
89                 return back()->with('error', 'Cette catégorie contient des dépenses : supprimez-les d’abord.');
90             }
91     
92             $categorieDepense->delete();
93     
94             return redirect()->route('admin.depenses.index')->with('success', 'La catégorie a été supprimée.');
95         }
96     }
