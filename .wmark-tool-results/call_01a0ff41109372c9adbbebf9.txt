1      <?php
2      
3      namespace App\Providers;
4      
5      use App\Models\ParametreSysteme;
6      use App\Models\AnneeScolaire;
7      use Illuminate\Support\ServiceProvider;
8      use Illuminate\Support\Facades\Schema;
9      use Illuminate\Support\Facades\View;
10     
11     class AppServiceProvider extends ServiceProvider
12     {
13         /**
14          * Register any application services.
15          */
16         public function register(): void
17         {
18             //
19         }
20     
21         /**
22          * Bootstrap any application services.
23          */
24         public function boot(): void
25         {
26             View::composer('*', static function ($view): void {
27                 $settings = request()->attributes->get('system_settings');
28                 if (! $settings) {
29                     $settings = Schema::hasTable('parametres_systeme')
30                         ? ParametreSysteme::query()->first()
31                         : null;
32                     $settings ??= ParametreSysteme::defaults();
33                     request()->attributes->set('system_settings', $settings);
34                 }
35     
36                 $view->with('systemSettings', $settings);
37     
38                 if (! request()->attributes->has('annee_en_cours')) {
39                     request()->attributes->set(
40                         'annee_en_cours',
41                         Schema::hasTable('annees_scolaires')
42                             ? AnneeScolaire::where('is_encours', true)->first()
43                             : null
44                     );
45                 }
46     
47                 $view->with('anneeEnCours', request()->attributes->get('annee_en_cours'));
48             });
49         }
50     }
