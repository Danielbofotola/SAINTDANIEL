120    test('the prefecture space is a dedicated page, not the shared placeholder', function () {
121        $prefet = User::where('role', 'prefetture')->first();
122    
123        $this->actingAs($prefet)
124            ->get(route('prefecture.dashboard'))
125            ->assertOk()
126            ->assertSee('Espace préfecture')
127            ->assertSee('Effectifs par cycle')
128            ->assertSee('Primaire')
