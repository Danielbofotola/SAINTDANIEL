1      <?php
2      
3      namespace App\Models;
4      
5      use Illuminate\Database\Eloquent\Builder;
6      use Illuminate\Database\Eloquent\Model;
7      
8      class Inscription extends Model
9      {
10         protected $fillable = ['eleve_id', 'classe_id', 'annee_scolaire_id', 'date_inscription', 'statut'];
11     
12         protected function casts(): array
13         {
14             return ['date_inscription' => 'date'];
15         }
16     
17         public function eleve()
18         {
19             return $this->belongsTo(Eleve::class);
20         }
21     
22         public function classe()
23         {
24             return $this->belongsTo(Classe::class);
25         }
26     
27         public function anneeScolaire()
28         {
29             return $this->belongsTo(AnneeScolaire::class);
30         }
31     
32         /**
33          * Limite la requête aux inscriptions de l'année scolaire en cours.
34          * Sans année en cours, aucune inscription ne correspond.
35          */
36         public function scopeAnneeEnCours(Builder $query): Builder
37         {
38             $anneeId = AnneeScolaire::enCours()?->id;
39     
40             return $anneeId
41                 ? $query->where('annee_scolaire_id', $anneeId)
42                 : $query->whereRaw('1 = 0');
43         }
44     }
