93         /**
94          * Décision de délibération de l'année précédente, par élève (clé = eleve_id).
95          *
96          * @param  array<int, int>  $eleveIds
97          * @return array<int, string>
98          */
99         private function origines(?AnneeScolaire $annee, array $eleveIds): array
100        {
101            if ($annee === null || $eleveIds === [] || ! Schema::hasTable('deliberations')) {
102                return [];
103            }
104    
105            $precedente = AnneeScolaire::precedente($annee);
106    
107            if ($precedente === null) {
108                return [];
109            }
110    
111            return Deliberation::query()
112                ->join('inscriptions', 'inscriptions.id', '=', 'deliberations.inscription_id')
113                ->where('deliberations.annee_scolaire_id', $precedente->id)
114                ->whereIn('inscriptions.eleve_id', $eleveIds)
115                ->pluck('deliberations.decision', 'inscriptions.eleve_id')
116                ->all();
117        }
