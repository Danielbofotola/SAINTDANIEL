96         {

98             $garcons = 0;
99             $filles = 0;

102                $decision = $ligne['decision'];
103                $decisions[$decision] ??= ['total' => 0, 'g' => 0, 'f' => 0];
104                $decisions[$decision]['total']++;
105

107                    'M' => $decisions[$decision]['g']++ + $garcons++,

109                    default => null,
110                };
111            }
112

114                $decisions[$cle] ??= ['total' => 0, 'g' => 0, 'f' => 0];
115            }
116    
117            return [
118                'total' => $lignes->count(),
119                'garcons' => $garcons,
120                'filles' => $filles,
121                'decisions' => $decisions,
122            ];
