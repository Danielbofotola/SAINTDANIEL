1      <?php
2      
3      namespace App\Models;
4      
5      use Illuminate\Database\Eloquent\Model;
6      use Illuminate\Support\Facades\DB;
7      use Illuminate\Support\Facades\Schema;
8      
9      class AnneeScolaire extends Model
10     {
11         /**
12          * Tables portant une clé étrangère vers annees_scolaires.
13          */
14         private const TABLES_LIEES = [
15             'inscriptions',
16             'periodes',
17             'affectations_cours',
18             'frais_scolaires',
19             'depenses',
20             'demandes_inscription',
21         ];
22     
23         protected $table = 'annees_scolaires';
24         protected $fillable = ['libelle', 'date_debut', 'date_fin', 'is_encours', 'statut'];
25     
26         protected function casts(): array
27         {
28             return [
29                 'date_debut' => 'date',
30                 'date_fin' => 'date',
31                 'is_encours' => 'boolean',
32             ];
33         }
34     
35         /**
36          * L'année scolaire « en cours » (unique), ou null si aucune n'est définie.
37          */
38         public static function enCours(): ?self
39         {
40             return static::where('is_encours', true)->first();
41         }
42     
43         /**
44          * L'année scolaire qui suit celle donnée (ou l'année en cours), d'après la
45          * date de début. Elle matérialise le passage : la délibération de N aboutit
46          * dans N+1. Renvoie null s'il n'existe aucune année postérieure.
47          */
48         public static function suivante(?self $annee = null): ?self
49         {
50             $annee ??= static::enCours();
51     
52             if ($annee === null) {
53                 return null;
54             }
55     
56             return static::query()
57                 ->where('statut', 'actif')
58                 ->where('date_debut', '>', $annee->date_debut)
59                 ->orderBy('date_debut')
60                 ->first();
61         }
62     
63         /**
64          * Les inscriptions rattachées à cette année scolaire.
65          */
66         public function inscriptions()
67         {
68             return $this->hasMany(Inscription::class);
69         }
70     
71         public function periodes()
72         {
73             return $this->hasMany(Periode::class);
74         }
75     
76         public function affectationsCours()
77         {
78             return $this->hasMany(AffectationCours::class);
79         }
80     
81         public function fraisScolaires()
82         {
83             return $this->hasMany(FraisScolaire::class);
84         }
85     
86         public function depenses()
87         {
88             return $this->hasMany(Depense::class);
89         }
90     
91         public function archive()
92         {
93             return $this->hasOne(ArchiveAnnee::class);
94         }
95     
96         /**
97          * Indique si l'année est référencée par des données métier
98          * (inscriptions, périodes, finances…), ce qui interdit sa suppression.
99          */
100        public function isUtilisee(): bool
101        {
102            foreach (self::TABLES_LIEES as $table) {
103                if (Schema::hasTable($table) && DB::table($table)->where('annee_scolaire_id', $this->id)->exists()) {
104                    return true;
105                }
106            }
107    
108            return false;
109        }
110    }
