116        <script>
117            (function () {
118                var payeParPaire = @json($soldePaye);
119                var montantsFrais = @json($montantsFrais);
120                var recherche = document.getElementById('eleveSearch');
121                var inscriptionSelect = document.getElementById('inscriptionSelect');
122                var fraisSelect = document.getElementById('fraisSelect');
123                var soldeInfo = document.getElementById('soldeInfo');
124    
125                function filtrerEleves() {
126                    if (!recherche || !inscriptionSelect) {
127                        return;
128                    }
129                    var terme = recherche.value.trim().toLowerCase();
130                    Array.prototype.forEach.call(inscriptionSelect.options, function (option) {
131                        if (!option.value) {
132                            option.hidden = false;
133                            return;
134                        }
135                        option.hidden = terme !== '' && option.text.toLowerCase().indexOf(terme) === -1;
136                    });
137                }
138    
139                function majSolde() {
140                    if (!inscriptionSelect || !fraisSelect || !soldeInfo) {
141                        return;
142                    }
143                    var inscription = inscriptionSelect.value;
144                    var frais = fraisSelect.value;
145                    if (!inscription || !frais) {
146                        soldeInfo.classList.add('d-none');
147                        return;
148                    }
149                    var exige = Number(montantsFrais[frais] || 0);
150                    var paye = Number(payeParPaire[inscription + '-' + frais] || 0);
151                    var reste = Math.max(exige - paye, 0);
152                    soldeInfo.classList.remove('d-none');
153                    soldeInfo.innerHTML = 'Montant exigé : <b>' + exige.toFixed(2) + '</b> · Déjà payé : <b>' + paye.toFixed(2) + '</b> · Reste à payer : <b>' + reste.toFixed(2) + '</b>';
154                }
155    
156                if (recherche) {
157                    recherche.addEventListener('input', filtrerEleves);
158                }
159                if (inscriptionSelect) {
160                    inscriptionSelect.addEventListener('change', majSolde);
161                }
162                if (fraisSelect) {
163                    fraisSelect.addEventListener('change', majSolde);
164                }
165            })();
166        </script>
