21         public function update(Request $request, Eleve $eleve): RedirectResponse
22         {
23             $data = $request->validate([
24                 'nom' => ['required', 'string', 'max:50'],
25                 'postnom' => ['required', 'string', 'max:50'],
26                 'prenom' => ['required', 'string', 'max:50'],
27                 'sexe' => ['required', 'in:M,F'],
28                 'date_naissance' => ['required', 'date', 'before:today'],
29                 'lieu_naissance' => ['nullable', 'string', 'max:100'],
30                 'statut' => ['required', 'in:actif,inactif,suspendu,exclu'],
31                 'photo' => ['nullable', 'image', 'max:2048'],
32                 'nom_mere' => ['required', 'string', 'max:100'],
33                 'telephone_mere' => ['required', 'string', 'max:20'],
34                 'profession_mere' => ['nullable', 'string', 'max:100'],
35                 'email_mere' => ['nullable', 'email', 'max:100'],
36                 'adresse_mere' => ['nullable', 'string'],
37                 'nom_pere' => ['nullable', 'string', 'max:100'],
38                 'telephone_pere' => ['nullable', 'string', 'max:20'],
39                 'profession_pere' => ['nullable', 'string', 'max:100'],
40                 'email_pere' => ['nullable', 'email', 'max:100'],
41                 'adresse_pere' => ['nullable', 'string'],
42                 'groupe_sanguin' => ['nullable', 'string', 'max:5'],
43                 'allergies_pathologies' => ['nullable', 'string'],
44                 'traitements_urgences' => ['nullable', 'string'],
45                 'medecin_referent' => ['nullable', 'string', 'max:150'],
46             ], $this->messages());
47     
48             DB::transaction(function () use ($request, $eleve, $data) {
49                 $photo = $request->file('photo')?->store('eleves/photos', 'public');
50                 $eleve->update(collect($data)->only(['nom', 'postnom', 'prenom', 'sexe', 'date_naissance', 'lieu_naissance', 'statut'])->merge($photo ? ['photo' => $photo] : [])->all());
51                 $eleve->tuteur()->update(collect($data)->only(['nom_mere', 'telephone_mere', 'profession_mere', 'email_mere', 'adresse_mere', 'nom_pere', 'telephone_pere', 'profession_pere', 'email_pere', 'adresse_pere'])->all());
52                 $eleve->sante()->updateOrCreate([], collect($data)->only(['groupe_sanguin', 'allergies_pathologies', 'traitements_urgences', 'medecin_referent'])->all());
53             });
54     
55             return redirect()->route('admin.eleves.show', $eleve)->with('success', 'Le profil de l’élève a été modifié.');
56         }
57     
58         public function storeDocument(Request $request, Eleve $eleve): RedirectResponse
59         {
60             $data = $request->validate([
61                 'type_document' => ['required', 'in:acte_naissance,bulletin,certificat_medical,autre'],
62                 'document' => ['required', 'file', 'mimes:pdf,jpg,jpeg,png', 'max:4096'],
63             ], [
64                 'document.required' => 'Veuillez sélectionner un fichier.',
65                 'document.max' => 'Le fichier ne doit pas dépasser 4 Mo.',
66                 'document.mimes' => 'Le fichier doit être au format PDF, JPG, JPEG ou PNG.',
67             ]);
68     
69             DocumentEleve::create([
70                 'eleve_id' => $eleve->id,
71                 'type_document' => $data['type_document'],
72                 'chemin_fichier' => $request->file('document')->store('eleves/documents', 'public'),
73                 'statut' => 'actif',
74                 'created_at' => now(),
75             ]);
76     
77             return back()->with('success', 'Le document a été ajouté au dossier.');
78         }
79     
80         public function show(Eleve $eleve): View
81         {
82             $eleve->load([
83                 'tuteur',
84                 'sante',
85                 'documents' => fn ($query) => $query->where('statut', 'actif')->latest('created_at'),
86             ]);
87     
88             // Seule l'inscription de l'année en cours est présentée sur le dossier.
89             $inscription = $eleve->inscriptions()
90                 ->anneeEnCours()
91                 ->where('statut', 'actif')
92                 ->with(['classe.cycle', 'classe.section', 'classe.option'])
93                 ->latest('date_inscription')
94                 ->first();
95     
96             return view('admin.eleves.show', compact('eleve', 'inscription'));
97         }
